    // =========================================================================
// lib/common/config.dart - ONDE está o servidor JSON?
// -------------------------------------------------------------------------
// A pergunta não se coloca para um aplicativo web servido pelo próprio
// servidor: as requisições vão para a MESMA origem da página
// (um endereço relativo, /api/...). Mas um aplicativo MOBILE não tem
// origem: é preciso informar a ele o endereço completo do servidor.
//   - o emulador Android enxerga o computador que o executa no
//     endereço 10.0.2.2 (localhost, para ele, é o próprio emulador);
//   - um celular de verdade, conectado à mesma rede Wi-Fi do computador, deve
//     usar o endereço do computador nessa rede (192.168.1.20...):
//     ele é informado quando o aplicativo é iniciado:
//         flutter run --dart-define=SERVER=http://192.168.1.20:8080 ...
//   - no desenvolvimento web (flutter run -d chrome), o servidor de
//     desenvolvimento do Flutter encaminha as requisições /api para o servidor JSON
//     (arquivo web_dev_config.yaml): um endereço relativo basta;
//   - em um computador (Windows, macOS, Linux): localhost.
// =========================================================================

import 'package:flutter/foundation.dart';

// o valor dado por --dart-define=SERVER=..., '' por padrão.
// String.fromEnvironment é avaliado em tempo de COMPILAÇÃO: é uma constante
const _forcedServer = String.fromEnvironment('SERVER');

class Config {
  // kIsWeb: true se o aplicativo roda em um navegador;
  // defaultTargetPlatform: o sistema (android, iOS, windows...)
  static String serverUrl = _forcedServer.isNotEmpty
      ? _forcedServer
      : kIsWeb
      ? '' // mesma origem: endereços relativos
      : defaultTargetPlatform == TargetPlatform.android
      ? 'http://10.0.2.2:8080'
      : 'http://localhost:8080';
}
    • linha 24: um endereço informado na inicialização da aplicação, por meio da opção [--dart-define]: [flutter run --dart-define=SERVER=http://192.168.1.20:8080 ...]. [String.fromEnvironment] é avaliado em tempo de compilação: é uma constante, “”, se a opção não for informada;
    • Linhas 29–35: caso contrário, um endereço padrão, dependendo da plataforma. [kIsWeb] é verdadeiro se a aplicação estiver sendo executada em um navegador; [defaultTargetPlatform] indica em qual sistema ela está sendo executada ([TargetPlatform.android], [TargetPlatform.windows]...):
      • na web: um endereço relativo (''), a mesma origem da aplicação;
      • no emulador [Android]: [http://10.0.2.2:8080]. O emulador é uma máquina independente, com sua própria rede: para o emulador, [localhost] refere-se ao próprio emulador. O endereço especial [10.0.2.2] refere-se ao computador que o executa;
      • em um computador ([Windows]...): [http://localhost:8080].

    Um celular real, conectado à mesma rede Wi-Fi do computador do servidor, deve usar o endereço do computador nessa rede (o comando [ipconfig] do [Windows] o fornece): ele é informado por meio de [--dart-define]. O firewall do computador também deve permitir conexões de entrada na porta 8080.

    // =========================================================================
// lib/ex22_http/main.dart - uma primeira requisição HTTP ao servidor
// -------------------------------------------------------------------------
// O servidor JSON do RdvMedecins deve estar iniciado (porta 8080). Pedimos a ele
// um captcha: GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - http.get(uri): a função do pacote http que envia uma
//     requisição GET. Ela retorna um Future da resposta (http.Response): seu código
//     de status (statusCode), seus cabeçalhos (headers), seu corpo (body);
//   - jsonDecode(text): o corpo JSON -> objetos Dart (Map, List...);
//   - o endereço do servidor depende de onde o aplicativo roda
//     (emulador Android, celular, navegador...): veja common/config.dart;
//   - três estados a exibir: em andamento, bem-sucedido, com falha;
//   - a imagem do captcha é SVG: o pacote flutter_svg sabe desenhá-la.
// =========================================================================

import 'dart:convert';

import 'package:flutter/material.dart';
import 'package:flutter_svg/flutter_svg.dart';
import 'package:http/http.dart' as http;

import '../common/config.dart';

void main() {
  runApp(
    MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const CaptchaPage(),
    ),
  );
}

class CaptchaPage extends StatefulWidget {
  const CaptchaPage({super.key});

  @override
  State<CaptchaPage> createState() => _CaptchaPageState();
}

class _CaptchaPageState extends State<CaptchaPage> {
  // o que GET /api/captcha retorna: um record
  ({String id, String svg})? _captcha;
  bool _loading = true;
  String? _error;
  int _requestCount = 0;
  int _requestNumber = 0; // o número da última requisição

  @override
  void initState() {
    super.initState();
    _load(); // a primeira requisição, assim que a tela aparece
  }

  Future<void> _load() async {
    final number = ++_requestNumber;
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      final response = await http.get(Uri.parse('${Config.serverUrl}/api/captcha'));
      // http.get NÃO lança exceção para uma resposta de erro (4xx,
      // 5xx): o código de status é testado
      if (response.statusCode != 200) {
        throw Exception('the server answered ${response.statusCode} ${response.reasonPhrase}');
      }
      final json = jsonDecode(response.body) as Map<String, dynamic>;
      // a resposta de uma requisição OBSOLETA (uma nova foi feita nesse meio-tempo), ou
      // uma que chegou depois que a tela foi fechada: ela é ignorada
      if (!mounted || number != _requestNumber) return;
      setState(() {
        _captcha = (id: json['id'] as String, svg: json['svg'] as String);
        _requestCount++;
        _loading = false;
      });
    } catch (e) {
      // nenhuma resposta (servidor parado, endereço errado...), ou um erro
      if (!mounted || number != _requestNumber) return;
      setState(() {
        _error = '$e';
        _captcha = null;
        _loading = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    final captcha = _captcha;
    return Scaffold(
      appBar: AppBar(title: const Text('Example 22 - a first HTTP request')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          Text(
            'Server: ${Config.serverUrl.isEmpty ? '(same origin)' : Config.serverUrl}',
            style: const TextStyle(color: Colors.grey),
          ),
          const SizedBox(height: 12),
          if (_loading) const LinearProgressIndicator(),
          if (_error != null)
            Card(
              color: Colors.red.shade50,
              child: Padding(
                padding: const EdgeInsets.all(12),
                child: Text(
                  'Unable to get a captcha: $_error.\n'
                  'Is the RdvMedecins JSON server started (port 8080)?',
                ),
              ),
            ),
          if (captcha != null && !_loading) ...[
            // SvgPicture.string: desenhar uma imagem SVG dada como texto
            Container(
              padding: const EdgeInsets.all(4),
              decoration: BoxDecoration(
                border: Border.all(color: Colors.grey.shade300),
                borderRadius: BorderRadius.circular(6),
              ),
              child: SvgPicture.string(captcha.svg, height: 60),
            ),
            const SizedBox(height: 8),
            Text('id: ${captcha.id}', style: const TextStyle(fontFamily: 'RobotoMono', fontSize: 11)),
          ],
          const SizedBox(height: 12),
          Align(
            alignment: Alignment.centerLeft,
            child: FilledButton(onPressed: _loading ? null : _load, child: const Text('Another image')),
          ),
          const SizedBox(height: 8),
          Text('$_requestCount successful request(s)', style: const TextStyle(color: Colors.grey)),
        ],
      ),
    );
  }
}

    // =========================================================================
// lib/common/http_client.dart - um cliente HTTP que gerencia COOKIES
// -------------------------------------------------------------------------
// O servidor RdvMedecins coloca o token JWT em um COOKIE (httpOnly).
// Em um navegador, é o navegador que guarda os cookies e os envia
// de volta a cada requisição: o código do aplicativo não tem nada a fazer.
// Em um celular, NÃO há navegador: cabe ao aplicativo
// desempenhar seu papel, ou seja:
//   1. ler o cabeçalho "Set-Cookie" de cada resposta e guardar os cookies;
//   2. enviá-los de volta no cabeçalho "Cookie" de cada requisição.
// Este cliente HTTP faz isso, ENVOLVENDO o cliente do pacote http
// (http.BaseClient: basta escrever o método send). Os cookies são
// também guardados no dispositivo (shared_preferences): continuamos logados
// quando paramos e reiniciamos o aplicativo.
// Na web, ele não faz nada além do cliente comum: o
// navegador proíbe o aplicativo de ler ou escrever esses cabeçalhos.
// =========================================================================

import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:http/http.dart' as http;
import 'package:shared_preferences/shared_preferences.dart';

class CookieClient extends http.BaseClient {
  static const _key = 'rdvmedecins.cookies';

  final http.Client _inner = http.Client();
  final SharedPreferences? _storage;
  // nome do cookie -> seu valor
  final Map<String, String> _cookies = {};

  CookieClient([this._storage]) {
    // os cookies guardados durante um uso anterior
    final stored = _storage?.getString(_key);
    if (stored != null) {
      _cookies.addAll((jsonDecode(stored) as Map<String, Object?>).cast<String, String>());
    }
  }

  @override
  Future<http.StreamedResponse> send(http.BaseRequest request) async {
    if (!kIsWeb && _cookies.isNotEmpty) {
      // "rdvmedecins_jwt=eyJhbGci...; other=value"
      request.headers['cookie'] = [for (final MapEntry(:key, :value) in _cookies.entries) '$key=$value'].join('; ');
    }
    final response = await _inner.send(request);
    final setCookie = response.headers['set-cookie'];
    if (!kIsWeb && setCookie != null) {
      _store(setCookie);
    }
    return response;
  }

  // Set-Cookie: rdvmedecins_jwt=eyJ...; Max-Age=7200; Path=/api; Expires=Tue, 29 Sep 2026 ...; HttpOnly
  // (vários cookies são separados por vírgulas). Guardamos apenas o
  // nome e o valor de cada cookie; um valor vazio (logout) o apaga
  void _store(String setCookie) {
    // um nome seguido de "=", no início do cabeçalho ou após uma vírgula
    for (final m in RegExp(r'(?:^|,)\s*([^=;,\s]+)=([^;]*)').allMatches(setCookie)) {
      final name = m[1]!;
      final value = m[2]!.trim();
      if (value.isEmpty) {
        _cookies.remove(name);
      } else {
        _cookies[name] = value;
      }
    }
    _storage?.setString(_key, jsonEncode(_cookies));
  }

  @override
  void close() => _inner.close();
}
    • Linha 25: o pacote [http] permite encapsular o seu cliente: você herda de [http.BaseClient] e só precisa implementar o método [send], por onde passam todas as requisições ([get], [post]...);
    • linhas 28–31: o cliente encapsulado, o armazenamento do dispositivo (opcional: [?]) e os cookies (nome → valor);
    • linhas 33–37: na criação, os cookies armazenados durante um uso anterior da aplicação. Isso mantém você logado quando fecha e reabre a aplicação: em um celular, equivale a um navegador que conserva seus cookies de uma visita para a outra;
    • linhas 42–52: cada requisição recebe o cabeçalho [Cookie] (“rdvmedecins_jwt=eyJhbGci...; other=value”) e, em seguida, o cabeçalho [Set-Cookie] da resposta é lido. [kIsWeb]: na web, a aplicação não faz nada além de um cliente comum, pois o navegador a impede de ler ou escrever esses cabeçalhos (ele cuida deles por conta própria);
    • linhas 58–69: o cabeçalho [Set-Cookie]. Vários cookies são separados por vírgulas, mas a data [Expires] em si contém uma vírgula (“Tue, 29 Sep 2026”): a expressão regular reconhece apenas um nome seguido de [=], seja no início do cabeçalho, seja depois de uma vírgula. Apenas o nome e o valor de cada cookie são mantidos; um valor vazio (logout: o servidor envia [rdvmedecins_jwt=; Expires=Thu, 01 Jan 1970...]) exclui o cookie. [_storage?.setString(...)]: a chamada só é feita se [_storage] for diferente de [null].

    Um gerenciamento de cookies mais completo (domínios, caminhos, datas de expiração) está disponível em pacotes ([cookie_jar], com o cliente [dio]); nossa implementação, de apenas algumas linhas, basta para UM servidor. Nota de segurança: aqui o token é armazenado em texto simples no dispositivo. Uma aplicação em produção usaria o pacote [flutter_secure_storage] (veja o exemplo 20).

    // =========================================================================
// lib/ex23_login/command.dart - uma AÇÃO, seu estado, seu resultado
// -------------------------------------------------------------------------
// Uma ação assíncrona (fazer login, salvar...) tem um estado que
// a tela deve mostrar: está EM ANDAMENTO (botão desabilitado, indicador)?
// terminou com um ERRO (mensagem)? É o padrão "Command"
// do guia de arquitetura do Flutter, reduzido ao essencial:
//
//     final login = Command();
//     login.execute(() => logIn(...));          // inicia a ação
//     login.running, login.error               // seu estado
//
// Um Command é um ChangeNotifier: a tela o escuta (ListenableBuilder)
// e se redesenha quando ele começa e quando termina.
// =========================================================================

import 'package:flutter/foundation.dart';

class Command extends ChangeNotifier {
  bool _running = false;
  Object? _error;

  bool get running => _running;
  Object? get error => _error;

  // executa a ação, se outra ainda não estiver em andamento
  Future<void> execute(Future<void> Function() action) async {
    if (_running) return; // sem envio duplo (toque duplo no botão)
    _running = true;
    _error = null;
    notifyListeners();
    try {
      await action();
    } catch (e) {
      _error = e; // o erro é MANTIDO, para ser exibido
    } finally {
      // finally: executado em TODOS os casos, sucesso ou erro
      _running = false;
      notifyListeners();
    }
  }

  void clearError() {
    _error = null;
    notifyListeners();
  }
}

class ErrorResponse implements Exception {

class _LoginPageState extends State<LoginPage> {
  Map<String, dynamic>? _user; // null: ninguém está logado
  List<dynamic> _doctors = [];
  Map<String, dynamic>? _captcha;
  bool _ready = false; // a verificação inicial terminou?
  String? _serverError;
  final _loginCommand = Command();
  final _login = TextEditingController(text: 'admin');
  final _password = TextEditingController();
  final _captchaAnswer = TextEditingController();

  void initState() {

  @override

    _loginCommand.dispose();

    _captchaAnswer.dispose();

  Uri _url(String path) => Uri.parse('${Config.serverUrl}/api$path');

  // uma requisição GET cujo corpo JSON queremos; uma exceção se não for 2xx
  Future<dynamic> _read(String path) async {
    final response = await widget.client.get(_url(path));
    if (response.statusCode >= 300) throw ErrorResponse(response);
    return jsonDecode(utf8.decode(response.bodyBytes));
  }

  // um POST com corpo JSON
  Future<http.Response> _post(String path, Object body) {
    return widget.client.post(
      _url(path),
      headers: {'content-type': 'application/json'}, // o corpo é JSON
      body: jsonEncode(body),
    );

    final captcha = await _read('/captcha') as Map<String, dynamic>;

    if (mounted) setState(() => _doctors = doctors);

    try {
      final user = await _read('/auth/me') as Map<String, dynamic>;
      setState(() => _user = user);
      await _loadDoctors();
    } on ErrorResponse catch (e) {
      // "on Type catch": captura SOMENTE este tipo de exceção
      if (e.response.statusCode == 401) {
        await _loadCaptcha();
      } else {
        _serverError = 'The server answered ${e.response.statusCode}.';
      }
    } catch (_) {
      _serverError = 'Is the RdvMedecins JSON server started (${Config.serverUrl})?';

// O JSON decodificado não é tipado: Map<String, dynamic>, List...

// do JSON recebido, com o "pattern matching" do Dart 3:
//
//   switch (json) {
//     {'login': String login, 'name': String name} => ...   // o formato esperado
//     _ => throw FormatException(...)                       // todo o resto
//   }
//
// O padrão {'login': String login, ...} verifica que json é um Map, que
// estas chaves existem, que seus valores têm o tipo certo, e armazena estes

enum Role { admin, doctor, user }

// "ADMIN" -> Role.admin
Role _role(String name) => Role.values.byName(name.toLowerCase());

// POST /api/auth/login, GET /api/auth/me
class User {
  final String login;
  final String name;
  final Role role;

  const User(this.login, this.name, this.role);

  factory User.fromJson(Object? json) => switch (json) {
    {'login': final String login, 'name': final String name, 'role': final String role} => User(
      login,
      name,
      _role(role),
    ),
    _ => throw FormatException('unexpected User: $json'),

}

class Person {
  final int id;
  final int version;
  final String title;
  final String firstName;
  final String lastName;
  final String? login; // somente para o ADMIN

  const Person(this.id, this.version, this.title, this.firstName, this.lastName, [this.login]);

  factory Person.fromJson(Object? json) => switch (json) {
    {
      'id': final int id,
      'version': final int version,
      'title': final String title,
      'firstName': final String firstName,
      'lastName': final String lastName,
    } =>
      // a chave "login" é opcional: ela é lida separadamente
      Person(id, version, title, firstName, lastName, (json as Map<String, Object?>)['login'] as String?),
    _ => throw FormatException('unexpected Person: $json'),
  };

  // uma lista JSON -> uma lista de pessoas
  static List<Person> list(Object? json) => [for (final p in json as List<Object?>) Person.fromJson(p)];

  String get fullName => '$title $firstName $lastName';
}

// (rede fora do ar, servidor parado, endereço errado), ou uma resposta que não
// vem do nosso servidor (uma página de erro de um proxy). Convertemos
// tudo isso em UMA só classe, ApiError, que carrega a chave de tradução
// enviada pelo servidor (e, se houver, seus parâmetros e o erro de
// cada campo).
// =========================================================================

import 'dart:convert';

import 'package:http/http.dart' as http;

// "implements Exception": um objeto que pode ser lançado (throw) e capturado (catch)
class ApiError implements Exception {
  final int statusCode; // 0: o servidor não respondeu
  final String key; // p. ex. 'ERRORS.LOGIN_TAKEN'
  final Map<String, String> params;
  final Map<String, String> fields;

  const ApiError(this.statusCode, this.key, {this.params = const {}, this.fields = const {}});

  // uma resposta de erro (4xx, 5xx) -> um ApiError
  factory ApiError.fromResponse(http.Response response) {
    Object? body;
    try {
      body = jsonDecode(utf8.decode(response.bodyBytes));
    } on FormatException {
      body = null; // não é JSON: não foi o nosso servidor que respondeu
    }
    // o "pattern matching" de um Map: se o corpo tem uma chave
    // "key" do tipo String, ela é armazenada na variável key
    if (body case {'key': final String key}) {
      final json = body as Map<String, Object?>;
      return ApiError(response.statusCode, key, params: _strings(json['params']), fields: _strings(json['fields']));
    }
    // sem key: foi o proxy ou a rede que respondeu no lugar
    return ApiError(
      response.statusCode,
      response.statusCode >= 500 ? 'ERRORS.SERVER_UNREACHABLE' : 'ERRORS.SERVER_ERROR',
    );
  }

  // um objeto JSON { "name": "ERRORS..." } -> Map<String, String> (vazio caso contrário)
  static Map<String, String> _strings(Object? json) =>
      json is Map<String, Object?> ? {for (final e in json.entries) e.key: '${e.value}'} : const {};

  @override
  String toString() => '$statusCode $key';
}

    Todas as respostas de erro do servidor têm o mesmo formato. No entanto, uma requisição também pode falhar DE OUTRO MODO: nenhuma resposta, ou uma resposta que não vem do nosso servidor. Convertemos tudo isso em UMA só classe, [ApiError].

    • linhas 19–23: um status (0: nenhuma resposta), uma chave de tradução, seus parâmetros e o erro de cada campo;
    • linhas 28–44: uma resposta de erro → um [ApiError]. [if (body case {'key': final String key})]: filtragem em um [if] (“if-case”): se o corpo tiver uma chave [key] do tipo string, ela é armazenada na variável [key]. Caso contrário, não foi o nosso servidor que respondeu;
    • linha 53: “401 LOGIN.ERROR”, para exibição rápida.

// -------------------------------------------------------------------------
// No exemplo 23, cada requisição repetia as mesmas etapas: endereço do
// servidor, prefixo /api, cabeçalho JSON, teste do código de status, leitura do
// corpo do erro... Elas são reunidas aqui:
//
//   final json = await api.get('/doctors');
//   await api.post('/appointments', {'day': day, 'slotId': 3});
//
//   - em caso de sucesso, o Future retorna o corpo JSON, DECODIFICADO (Maps,
//     Lists, Strings, números...);
//   - em caso de falha, ele é concluído com um ApiError.
// Todas as requisições passam pelo método request: é o lugar onde
// um problema COMUM a todas elas seria tratado (veja o estudo de caso e a

import 'dart:convert';

import 'package:http/http.dart' as http;

import 'config.dart';
import 'api_error.dart';

class Api {
  final http.Client _client;

  Api(this._client);

  Future<Object?> request(String method, String url, [Object? body]) async {
    final request = http.Request(method, Uri.parse('${Config.serverUrl}/api$url'));
    if (body != null) {
      // um corpo JSON (o servidor recusa um POST que não tenha um)
      request.headers['content-type'] = 'application/json';
      request.body = jsonEncode(body);
    }
    final http.Response response;
    try {
      // send envia a requisição; fromStream espera o corpo completo da resposta
      response = await http.Response.fromStream(await _client.send(request));
    } catch (_) {
      // NENHUMA resposta (rede, servidor parado, endereço errado)
      throw const ApiError(0, 'ERRORS.SERVER_UNREACHABLE');
    }
    if (response.statusCode >= 400) {
      throw ApiError.fromResponse(response);
    }
    return response.bodyBytes.isEmpty ? null : jsonDecode(utf8.decode(response.bodyBytes));
  }

  Future<Object?> get(String url) => request('GET', url);
  // um POST sempre tem um corpo JSON: {} por padrão
  Future<Object?> post(String url, [Object body = const {}]) => request('POST', url, body);
  Future<Object?> put(String url, Object body) => request('PUT', url, body);
  Future<Object?> delete(String url, [Object? body]) => request('DELETE', url, body);
}
    • Linhas 31–49: TODAS as requisições passam por este método: o endereço do servidor e o prefixo [/api], o corpo em JSON, o envio ([_client.send], seguido de [http.Response.fromStream], que espera o corpo completo), os erros convertidos em [ApiError] e o corpo da resposta decodificado. É aqui que trataríamos um problema comum a todas as requisições (veja o estudo de caso e a resposta 401);
    • linhas 52–56: quatro atalhos. Um POST sempre tem um corpo em JSON: [{}] por padrão.

  Future<List<Person>> getClients() async => Person.list(await _api.get('/clients'));

  // Uri monta "doctorId=1&day=2030-01-15", codificando o que for necessário
  Future<Schedule> getSchedule(int doctorId, String day) async {

}

// lib/ex24_api_layer/schedule_model.dart - o "view model" da tela
// -------------------------------------------------------------------------
// O guia de arquitetura do Flutter recomenda separar:
//   - a VIEW: os widgets, que exibem e repassam as ações do
//     usuário;
//   - o VIEW MODEL (ViewModel): um ChangeNotifier que mantém o estado
//     da tela (o médico escolhido, o dia, a agenda recebida...), as ações
//     (escolher, marcar, cancelar), e que chama a camada de acesso à
//     API.
// A view não tem mais nenhuma lógica: é simples de ler; o view model
// não tem widgets: é simples de testar.
//
// Ele também faz o que não queremos reescrever em cada tela:
//   - um CACHE: a agenda de cada (médico, dia) já lida é mantida; voltar

import 'package:flutter/foundation.dart';

import '../common/api_error.dart';
import '../common/types.dart';
import 'rdvmedecins.dart';

String _twoDigits(int n) => n.toString().padLeft(2, '0');
String formatDay(DateTime d) => '${d.year}-${_twoDigits(d.month)}-${_twoDigits(d.day)}';

class ScheduleModel extends ChangeNotifier {
  final RdvMedecins _rdvmedecins;

  ScheduleModel(this._rdvmedecins);

  // o estado
  User? user;
  List<Person> doctors = [];
  List<Person> clients = [];
  int? doctorId;
  String day = formatDay(DateTime.now().add(const Duration(days: 1))); // amanhã
  int? clientId;
  Schedule? schedule;
  bool loading = false;
  ({bool success, String text})? message;

  final Map<String, Schedule> _cache = {};
  int _readNumber = 0;

  // valores calculados
  bool get loggedIn => user != null;
  bool get isManager => loggedIn && user!.role != Role.user;

  // as ações
  Future<void> changeUser(User? u) async {
    user = u;
    doctors = [];
    clients = [];
    schedule = null;
    doctorId = null;
    _cache.clear(); // os direitos mudaram: tudo é esquecido
    notifyListeners();
    if (u == null) return;
    doctors = await _rdvmedecins.getDoctors();
    if (isManager) clients = await _rdvmedecins.getClients(); // um USER receberia 403
    notifyListeners();
  }

  void chooseDoctor(int id) {
    doctorId = id;
    message = null;
    _readSchedule();
  }

  void chooseDay(DateTime d) {
    day = formatDay(d);
    message = null;
    _readSchedule();
  }

  void chooseClient(int? id) {
    clientId = id;
    notifyListeners();
  }

  Future<void> book(ScheduleSlot c) =>
      _act(() => _rdvmedecins.book(day, c.id, isManager ? clientId : null), 'Slot ${c.timeRange} booked.');

  Future<void> cancel(ScheduleSlot c) =>
      _act(() => _rdvmedecins.cancelAppointment(c.appointmentId!), 'Appointment of ${c.timeRange} cancelled.');

  // uma ação, e SEMPRE uma nova leitura da agenda: o SERVIDOR é a referência
  Future<void> _act(Future<void> Function() action, String success) async {
    try {
      await action();
      message = (success: true, text: success);
    } on ApiError catch (e) {
      message = (success: false, text: '$e');
    }
    await _readSchedule();
  }

  Future<void> _readSchedule() async {
    final id = doctorId;
    if (id == null) return;
    final key = '$id/$day';
    final number = ++_readNumber;
    // primeiro o cache (ou nada), depois a leitura
    schedule = _cache[key];
    loading = true;
    notifyListeners();
    try {
      final fetched = await _rdvmedecins.getSchedule(id, day);
      if (number != _readNumber) return; // uma leitura mais recente foi iniciada
      schedule = _cache[key] = fetched;
    } on ApiError catch (e) {
      if (number != _readNumber) return;
      schedule = null;
      message = (success: false, text: '$e');
    }
    loading = false;
    notifyListeners();
  }
}

A view não contém mais nenhuma lógica; a view model não tem widgets: pode ser testada sem uma tela.

    • linhas 35–43: o estado da tela;
    • linhas 45–46: um cache: a agenda de cada (médico, dia) que já foi lida é armazenada; ao voltar a ela, ela é exibida imediatamente enquanto é relida. E o número da última leitura, para ignorar respostas desatualizadas (veja o exemplo 19). Isso, em poucas linhas, é o que a biblioteca [TanStack Query] do curso de [React] faz;
    • linhas 49–50: valores calculados;

// ApiError por common/api.dart):

//   400 { key: 'ERRORS.INVALID_FORM', fields: { lastName: 'ERRORS.LASTNAME_REQUIRED' } }

//   409 { key: 'ERRORS.CONCURRENT_UPDATE' }         (bloqueio otimista)

// O erro de CADA campo (ApiError.fields) é armazenado no estado e

// Experimente: um sobrenome vazio, um título "Dr", um homônimo (Alice DURAND), um

enum Field { title, firstName, lastName, login, password }

class ClientsPage extends StatefulWidget {
  const ClientsPage({super.key});

  @override
  State<ClientsPage> createState() => _ClientsPageState();
}

class _ClientsPageState extends State<ClientsPage> {
  late final Api _api = context.read<Api>();
  User? _user;
  List<Person> _clients = [];

  // o formulário: _editedId == null -> criação, caso contrário atualização
  int? _editedId;
  int _version = 0;
  String _title = 'Mr';
  final _texts = {
    for (final c in [Field.firstName, Field.lastName, Field.login, Field.password]) c: TextEditingController(),
  };
  // o erro de cada campo (uma CHAVE de tradução), devolvido pelo servidor
  Map<String, String> _errors = {};
  // a mensagem geral do último erro, e a de um sucesso (chave + parâmetros)
  ApiError? _pageError;
  ({String key, Map<String, String> params})? _success;
  // envio ao servidor: um Command (exemplo 23), que informa se está em andamento
  final _sending = Command();

  bool get _isAdmin => _user?.role == Role.admin;

  @override
  void dispose() {
    for (final c in _texts.values) {
      c.dispose();
    }
    _sending.dispose();
    super.dispose();
  }

  Future<void> _reload() async {
    final clients = Person.list(await _api.get('/clients'));
    if (mounted) setState(() => _clients = clients);
  }

  Future<void> _onUserChange(User? u) async {
    setState(() {
      _user = u;
      _clients = [];
    });
    if (u?.role == Role.admin) await _reload();
  }

  void _clearMessages() {
    _errors = {};
    _pageError = null;
    _success = null;
  }

  void _newForm() {
    setState(() {
      _editedId = null;
      _version = 0;
      _title = 'Mr';
      for (final c in _texts.values) {
        c.clear();
      }
      _clearMessages();
    });
  }

  void _edit(Person c) {
    setState(() {
      _editedId = c.id;
      // a VERSÃO lida com o registro volta com a atualização: se alguém
      // modificou o registro nesse meio tempo, o servidor responderá 409 CONCURRENT_UPDATE
      _version = c.version;
      _title = c.title;
      _texts[Field.firstName]!.text = c.firstName;
      _texts[Field.lastName]!.text = c.lastName;
      _texts[Field.login]!.text = c.login ?? '';
      _texts[Field.password]!.clear();
      _clearMessages();
    });
  }

  Future<void> _save() async {
    setState(_clearMessages);
    final input = {
      'title': _title,
      for (final MapEntry(key: field, value: c) in _texts.entries) field.name: c.text,
      'version': _version,
    };
    try {
      final client = Person.fromJson(
        _editedId == null ? await _api.post('/clients', input) : await _api.put('/clients/$_editedId', input),
      );
      final key = _editedId == null ? 'FORM.ADDED' : 'FORM.UPDATED';
      _newForm(); // formulário limpo, mensagens limpas...
      // ... exceto esta
      setState(
        () => _success = (key: key, params: {'name': '${client.firstName} ${client.lastName}', 'login': client.login ?? ''}),
      );
      await _reload();
    } on ApiError catch (e) {
      setState(() {
        _pageError = e; // a mensagem geral
        _errors = e.fields; // o erro de cada campo
      });
    }
  }

  Future<void> _delete(Person c) async {
    final t = context.read<Translations>().t;
    final name = '${c.firstName} ${c.lastName}';
    final yes = await confirm(context, t('FORM.DELETE_CONFIRM', {'name': name}));
    if (!yes || !mounted) return;
    setState(_clearMessages);
    try {
      await _api.delete('/clients/${c.id}');
      setState(() => _success = (key: 'FORM.DELETED', params: {'name': name}));
    } on ApiError catch (e) {
      setState(() => _pageError = e);
    }
    await _reload();
  }

  @override
  Widget build(BuildContext context) {
    final translations = context.watch<Translations>();
    final t = translations.t;

    // um campo de entrada, com seu erro (uma chave do servidor) traduzido
    Widget field(Field c, {bool secret = false}) => TextField(
      controller: _texts[c],
      obscureText: secret,
      decoration: InputDecoration(
        labelText: t('FORM.${c.name.toUpperCase()}'),
        errorText: _errors[c.name] == null ? null : t(_errors[c.name]!),
        errorMaxLines: 2,
      ),
      // um campo modificado: seu erro desaparece
      onChanged: (_) {
        if (_errors.containsKey(c.name)) setState(() => _errors = {..._errors}..remove(c.name));
      },
    );

    final success = _success;
    final pageError = _pageError;
    return Scaffold(
      appBar: AppBar(
        title: Text(t('FORM.PAGE_TITLE')),
        actions: [
          // o idioma: um menu
          PopupMenuButton<String>(
            icon: const Icon(Icons.translate),
            onSelected: translations.changeLanguage,
            itemBuilder: (_) => [for (final l in languages) PopupMenuItem(value: l, child: Text(l.toUpperCase()))],
          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.all(12),
        children: [
          LoginBar(onUserChange: _onUserChange),
          if (_user != null && !_isAdmin)
            Card(
              color: Colors.amber.shade50,
              child: Padding(padding: const EdgeInsets.all(12), child: Text(t('FORM.ADMIN_ONLY'))),
            ),
          if (_isAdmin) ...[
            for (final c in _clients)
              ListTile(
                dense: true,
                title: Text(c.fullName),
                subtitle: Text(c.login ?? ''),
                trailing: Row(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    IconButton(tooltip: t('FORM.EDIT'), icon: const Icon(Icons.edit), onPressed: () => _edit(c)),
                    IconButton(
                      tooltip: t('FORM.DELETE'),
                      icon: const Icon(Icons.delete_outline, color: Colors.red),
                      onPressed: () => _delete(c),
                    ),
                  ],
                ),
              ),
            Card(
              child: Padding(
                padding: const EdgeInsets.all(12),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      t(_editedId == null ? 'FORM.TITLE_ADD' : 'FORM.TITLE_EDIT'),
                      style: Theme.of(context).textTheme.titleMedium,
                    ),
                    // um título "Dr" que o servidor recusará: para ver ERRORS.INVALID_TITLE
                    DropdownButton<String>(
                      value: _title,
                      items: [
                        for (final title in ['Mr', 'Mme', 'Melle', 'Dr'])
                          DropdownMenuItem(value: title, child: Text(title)),
                      ],
                      onChanged: (v) => setState(() {
                        _title = v!;
                        _errors = {..._errors}..remove('title');
                      }),
                    ),
                    if (_errors['title'] != null)
                      Text(t(_errors['title']!), style: const TextStyle(color: Colors.red, fontSize: 12)),
                    field(Field.firstName),
                    field(Field.lastName),
                    field(Field.login),
                    field(Field.password, secret: true),
                    const SizedBox(height: 12),
                    ListenableBuilder(
                      listenable: _sending,
                      builder: (context, _) => Wrap(
                        spacing: 8,
                        children: [
                          FilledButton(
                            onPressed: _sending.running ? null : () => _sending.execute(_save),
                            child: Text(t('FORM.SAVE')),
                          ),
                          OutlinedButton(onPressed: _newForm, child: Text(t('FORM.CANCEL'))),
                        ],
                      ),
                    ),
                    if (success != null)
                      Padding(
                        padding: const EdgeInsets.only(top: 12),
                        child: Text(t(success.key, success.params), style: const TextStyle(color: Colors.green)),
                      ),
                    // a mensagem geral, e o status HTTP, para entender
                    if (pageError != null)
                      Padding(
                        padding: const EdgeInsets.only(top: 12),
                        child: Text.rich(
                          TextSpan(
                            children: [
                              TextSpan(
                                text: t(pageError.key, pageError.params),
                                style: const TextStyle(color: Colors.red),
                              ),
                              TextSpan(
                                text: '  ($pageError)',
                                style: const TextStyle(color: Colors.grey, fontSize: 12),
                              ),
                            ],
                          ),
                        ),
                      ),
                  ],
                ),
              ),
            ),
          ],
        ],
      ),
    );
  }
}
